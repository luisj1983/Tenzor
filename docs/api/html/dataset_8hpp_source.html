139class ConcatDataset : public Dataset {
140public:
145 explicit ConcatDataset(std::vector<std::shared_ptr<Dataset>> datasets)
146 : datasets_(std::move(datasets)) {
147 if (datasets_.empty()) {
148 throw std::invalid_argument("Cannot create empty ConcatDataset");
149 }
150
151 // Precompute cumulative sizes
152 cumulative_sizes_.push_back(0);
153 for (const auto& dataset : datasets_) {
154 cumulative_sizes_.push_back(cumulative_sizes_.back() + dataset->size());
155 }
156 }
157
158 auto size() const -> size_t override {
159 return cumulative_sizes_.back();
160 }
161
162 auto get(size_t index) -> std::pair<Tensor, Tensor> override {
163 if (index >= size()) {
164 throw std::out_of_range("Dataset index out of range");
165 }
166
167 // Binary search to find which dataset contains this index
168 auto it = std::upper_bound(cumulative_sizes_.begin(), cumulative_sizes_.end(), index);
169 size_t dataset_idx = std::distance(cumulative_sizes_.begin(), it) - 1;
170 size_t local_idx = index - cumulative_sizes_[dataset_idx];
171
172 return datasets_[dataset_idx]->get(local_idx);
173 }
174
175private:
176 std::vector<std::shared_ptr<Dataset>> datasets_;
177 std::vector<size_t> cumulative_sizes_;

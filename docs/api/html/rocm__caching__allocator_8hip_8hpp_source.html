25struct Block {
26 void* ptr; // Device pointer
27 size_t size; // Block size in bytes
28 bool allocated; // Whether block is currently allocated
29 int device; // HIP device ID
30 hipStream_t stream; // Associated stream (for async operations)
31 size_t alignment; // Block alignment for HBM optimization
32
33 Block(void* p, size_t s, int dev, hipStream_t str = nullptr, size_t align = 256)
34 : ptr(p), size(s), allocated(false), device(dev), stream(str), alignment(align) {}
35
36 // Comparison for ordered containers (by size, then by pointer)
37 bool operator<(const Block& other) const {
38 if (size != other.size) return size < other.size;
39 return ptr < other.ptr;
40 }
41};

46struct BlockComparator {
47 bool operator()(const Block* a, const Block* b) const {
48 if (a->size != b->size) return a->size < b->size;

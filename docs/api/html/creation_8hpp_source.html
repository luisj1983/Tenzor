211template<typename T>
212auto from_data(const T* data, std::vector<int64_t> shape,
213 Device device = Device::cpu()) -> Tensor {
214 // Determine dtype from T
215 DType dtype;
216 if constexpr (std::is_same_v<T, float>) {
217 dtype = DType::Float32;
218 } else if constexpr (std::is_same_v<T, double>) {
219 dtype = DType::Float64;
220 } else if constexpr (std::is_same_v<T, int32_t>) {
221 dtype = DType::Int32;
222 } else if constexpr (std::is_same_v<T, int64_t>) {
223 dtype = DType::Int64;
224 } else if constexpr (std::is_same_v<T, uint8_t>) {
225 dtype = DType::UInt8;
226 } else if constexpr (std::is_same_v<T, bool>) {
227 dtype = DType::Bool;
228 } else {
229 static_assert(std::is_same_v<T, float>, "Unsupported type for from_data");
230 }
231
232 // Create empty tensor
233 auto tensor = empty(shape, dtype, device);
234
235 // Copy data
236 if (tensor.impl() && tensor.impl()->storage) {
237 size_t bytes = tensor.numel() * sizeof(T);
238
239 // For CPU device, use memcpy
240 if (device.type == Device::Type::CPU) {
241 std::memcpy(tensor.impl()->storage->data(), data, bytes);
242 } else {
243 // For GPU devices, use backend's copy method
244 auto* backend = tenzor::backend_registry().get_backend(device.type);
245 if (backend) {
246 backend->copy(tensor.impl()->storage->data(),
247 const_cast<T*>(data),
248 bytes,
249 tenzor::CopyKind::HostToDevice);
250 } else {
251 throw std::runtime_error("Backend not available for device type");
252 }
253 }
254 }
255
256 return tensor;
257}

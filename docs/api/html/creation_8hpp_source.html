185template<typename T>
186auto from_data(const T* data, std::vector<int64_t> shape,
187 Device device = Device::cpu()) -> Tensor {
188 // Determine dtype from T
189 DType dtype;
190 if constexpr (std::is_same_v<T, float>) {
191 dtype = DType::Float32;
192 } else if constexpr (std::is_same_v<T, double>) {
193 dtype = DType::Float64;
194 } else if constexpr (std::is_same_v<T, int32_t>) {
195 dtype = DType::Int32;
196 } else if constexpr (std::is_same_v<T, int64_t>) {
197 dtype = DType::Int64;
198 } else if constexpr (std::is_same_v<T, uint8_t>) {
199 dtype = DType::UInt8;
200 } else if constexpr (std::is_same_v<T, bool>) {
201 dtype = DType::Bool;
202 } else {
203 static_assert(std::is_same_v<T, float>, "Unsupported type for from_data");
204 }
205
206 // Create empty tensor
207 auto tensor = empty(shape, dtype, device);
208
209 // Copy data
210 if (tensor.impl() && tensor.impl()->storage) {
211 size_t bytes = tensor.numel() * sizeof(T);
212 std::memcpy(tensor.impl()->storage->data(), data, bytes);
213 }
214
215 return tensor;
216}

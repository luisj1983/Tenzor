61// Create tensor from data
62template<typename T>
63auto from_data(const T* data, std::vector<int64_t> shape,
64 Device device = Device::cpu()) -> Tensor {
65 // Determine dtype from T
66 DType dtype;
67 if constexpr (std::is_same_v<T, float>) {
68 dtype = DType::Float32;
69 } else if constexpr (std::is_same_v<T, double>) {
70 dtype = DType::Float64;
71 } else if constexpr (std::is_same_v<T, int32_t>) {
72 dtype = DType::Int32;
73 } else if constexpr (std::is_same_v<T, int64_t>) {
74 dtype = DType::Int64;
75 } else if constexpr (std::is_same_v<T, uint8_t>) {
76 dtype = DType::UInt8;
77 } else if constexpr (std::is_same_v<T, bool>) {
78 dtype = DType::Bool;
79 } else {
80 static_assert(std::is_same_v<T, float>, "Unsupported type for from_data");
81 }
82
83 // Create empty tensor
84 auto tensor = empty(shape, dtype, device);
85
86 // Copy data
87 if (tensor.impl() && tensor.impl()->storage) {
88 size_t bytes = tensor.numel() * sizeof(T);
89 std::memcpy(tensor.impl()->storage->data(), data, bytes);
90 }
91
92 return tensor;
93}

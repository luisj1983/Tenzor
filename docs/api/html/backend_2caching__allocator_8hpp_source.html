24struct Block {

26 size_t size; // Block size in bytes

31 Block(void* p, size_t s, int dev, cudaStream_t str = nullptr)
32 : ptr(p), size(s), allocated(false), device(dev), stream(str) {}
33
34 // Comparison for ordered containers (by size, then by pointer)
35 bool operator<(const Block& other) const {
36 if (size != other.size) return size < other.size;
37 return ptr < other.ptr;

44struct BlockComparator {

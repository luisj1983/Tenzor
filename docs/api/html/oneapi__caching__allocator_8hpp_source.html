20struct OneAPIBlock {
21 void* ptr; // USM pointer
22 size_t size; // Block size in bytes
23 bool allocated; // Whether block is currently allocated
24 int device; // Device index
25 bool is_shared; // Whether this is shared memory (vs device-only)
26
27 OneAPIBlock(void* p, size_t s, int dev, bool shared = true)
28 : ptr(p), size(s), allocated(false), device(dev), is_shared(shared) {}
29
30 // Comparison for ordered containers (by size, then by pointer)
31 bool operator<(const OneAPIBlock& other) const {
32 if (size != other.size) return size < other.size;
33 return ptr < other.ptr;
34 }
35};

40struct OneAPIBlockComparator {
41 bool operator()(const OneAPIBlock* a, const OneAPIBlock* b) const {
42 if (a->size != b->size) return a->size < b->size;
43 return a->ptr < b->ptr;
44 }
45};

313class OneAPICachedMemoryGuard {
314public:
315 OneAPICachedMemoryGuard(size_t size, int device = 0, bool shared = true)
316 : ptr_(nullptr), device_(device), size_(size) {
317 if (shared) {
318 ptr_ = OneAPICachingAllocator::get().allocate_shared(size, device);
319 } else {
320 ptr_ = OneAPICachingAllocator::get().allocate_device(size, device);
321 }
322 }
323
324 ~OneAPICachedMemoryGuard() {
325 if (ptr_) {
326 OneAPICachingAllocator::get().free(ptr_, device_);
327 }
328 }
329
330 // Move semantics
331 OneAPICachedMemoryGuard(OneAPICachedMemoryGuard&& other) noexcept
332 : ptr_(other.ptr_), device_(other.device_), size_(other.size_) {
333 other.ptr_ = nullptr;
334 }
335
336 OneAPICachedMemoryGuard& operator=(OneAPICachedMemoryGuard&& other) noexcept {
337 if (this != &other) {
338 if (ptr_) {
339 OneAPICachingAllocator::get().free(ptr_, device_);
340 }
341 ptr_ = other.ptr_;
342 device_ = other.device_;
343 size_ = other.size_;
344 other.ptr_ = nullptr;
345 }
346 return *this;
347 }

351 OneAPICachedMemoryGuard& operator=(const OneAPICachedMemoryGuard&) = delete;

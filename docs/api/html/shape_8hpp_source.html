171inline auto compute_strides(std::span<const int64_t> shape) -> std::vector<int64_t> {
172 std::vector<int64_t> strides(shape.size());
173 if (shape.empty()) return strides;
174
175 strides.back() = 1;
176 for (int64_t i = static_cast<int64_t>(shape.size()) - 2; i >= 0; --i) {
177 strides[i] = strides[i + 1] * shape[i + 1];
178 }

209inline auto broadcast_shapes(std::span<const int64_t> shape1,
210 std::span<const int64_t> shape2)
211 -> std::vector<int64_t> {
212 size_t ndim = std::max(shape1.size(), shape2.size());
213 std::vector<int64_t> result(ndim);
214
215 for (size_t i = 0; i < ndim; ++i) {
216 int64_t dim1 = i < shape1.size() ? shape1[shape1.size() - 1 - i] : 1;
217 int64_t dim2 = i < shape2.size() ? shape2[shape2.size() - 1 - i] : 1;
218
219 if (dim1 == dim2 || dim1 == 1 || dim2 == 1) {
220 result[ndim - 1 - i] = std::max(dim1, dim2);
221 } else {
222 throw std::runtime_error("Shapes are not broadcastable");
223 }
224 }
225
226 return result;
227}

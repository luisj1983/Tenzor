18struct VulkanBlock {
19 VkBuffer buffer; // Vulkan buffer handle
20 VkDeviceMemory memory; // Device memory handle
21 void* mapped_ptr; // Host-visible mapped pointer (or synthetic address)
22 size_t size; // Block size in bytes
23 bool allocated; // Whether block is currently allocated
24 bool is_host_visible; // Whether memory is host-visible (mappable)
25 int device; // Device index
26 uint32_t memory_type_index; // Memory type index for this block
27
28 VulkanBlock(VkBuffer buf, VkDeviceMemory mem, void* ptr, size_t s, int dev, uint32_t mem_type)
29 : buffer(buf), memory(mem), mapped_ptr(ptr), size(s),
30 allocated(false), is_host_visible(true), device(dev), memory_type_index(mem_type) {}
31
32 // Comparison for ordered containers (by size, then by pointer)
33 bool operator<(const VulkanBlock& other) const {
34 if (size != other.size) return size < other.size;
35 return mapped_ptr < other.mapped_ptr;
36 }
37};

234template<typename F, typename... Args>
235auto async_execute(F&& func, Args&&... args) -> Future<Tensor> {
236 // Create promise and future
237 auto promise = std::make_shared<Promise<Tensor>>();
238 auto future = Future<Tensor>(promise->get_state());
239
240 // Submit to thread pool
241 thread_pool().submit([promise, func = std::forward<F>(func),
242 ... captured_args = std::forward<Args>(args)]() mutable {
243 try {
244 // Execute function
245 Tensor result = func(std::forward<Args>(captured_args)...);
246 promise->set_value(std::move(result));
247 } catch (...) {
248 // Propagate exception
249 promise->set_exception(std::current_exception());
250 }
251 });
252
253 return future;
254}

343template<typename Op>
344auto async_gpu_op(const Device& device, Op&& op) -> Future<Tensor> {
345 auto promise = std::make_shared<Promise<Tensor>>();
346 auto future = Future<Tensor>(promise->get_state());
347
348 // Get stream for this device
349 StreamHandle stream = StreamManager::instance().get_stream(device);
350
351 // Submit operation on stream
352 thread_pool().submit([promise, stream, device, op = std::forward<Op>(op)]() mutable {
353 try {
354 // Execute on stream
355 Tensor result = op(stream);
356
357 // Synchronize stream to ensure completion
358 StreamManager::instance().synchronize_stream(stream);
359
360 promise->set_value(std::move(result));
361 } catch (...) {
362 promise->set_exception(std::current_exception());
363 }
364 });

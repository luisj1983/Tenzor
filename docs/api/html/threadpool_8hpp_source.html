121auto ThreadPool::submit(F&& func, Args&&... args) -> std::future<std::invoke_result_t<F, Args...>> {
122 using return_type = std::invoke_result_t<F, Args...>;
123
124 auto task = std::make_shared<std::packaged_task<return_type()>>(
125 std::bind(std::forward<F>(func), std::forward<Args>(args)...)
126 );
127
128 std::future<return_type> result = task->get_future();
129 {
130 std::unique_lock lock(queue_mutex_);
131 if (stop_) {
132 throw std::runtime_error("Cannot submit task to stopped thread pool");
133 }
134 tasks_.emplace([task]() { (*task)(); });
135 }
136 condition_.notify_one();
137 return result;
138}

141auto ThreadPool::parallel_for(int64_t begin, int64_t end, F&& func) -> void {
142 if (begin >= end) return;
143
144 const size_t num_tasks = std::min<size_t>(end - begin, num_threads_ * 4);
145 const int64_t chunk_size = (end - begin + num_tasks - 1) / num_tasks;
146
147 std::vector<std::future<void>> futures;
148 futures.reserve(num_tasks);
149
150 for (size_t i = 0; i < num_tasks; ++i) {
151 int64_t start = begin + i * chunk_size;
152 int64_t finish = std::min(start + chunk_size, end);
153
154 futures.push_back(submit([&func, start, finish]() {
155 for (int64_t j = start; j < finish; ++j) {
156 func(j);
157 }
158 }));
159 }
160
161 for (auto& future : futures) {
162 future.wait();
163 }
164}
